
-- -----------------------------------------------------
-- Schema banco_sangue
-- -----------------------------------------------------
DROP SCHEMA IF EXISTS banco_sangue ;

-- -----------------------------------------------------
-- Schema banco_sangue
-- -----------------------------------------------------
CREATE SCHEMA IF NOT EXISTS banco_sangue DEFAULT CHARACTER SET UTF8MB4 ;
USE banco_sangue ;

-- -----------------------------------------------------
-- Table TpSang
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS TpSang (
  idTpSang INT NOT NULL COMMENT 'Código do Tipo Sanguíneo',
  dscTpSang VARCHAR(2) NOT NULL COMMENT 'Descrição do Tipo Sanguíneo',
  RhTpSang CHAR(1) NOT NULL COMMENT 'RH do Tipo Sanguíneo',
  PRIMARY KEY (idTpSang))
ENGINE = InnoDB
DEFAULT CHARACTER SET = UTF8MB4;


-- -----------------------------------------------------
-- Table UF
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS UF (
  idUF CHAR(2) NOT NULL,
  dscUF VARCHAR(50) NOT NULL,
  PRIMARY KEY (idUF))
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table Cidade
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS Cidade (
  idCidade INT NOT NULL,
  dscCidade VARCHAR(45) NOT NULL,
  idUF CHAR(2) NOT NULL,
  PRIMARY KEY (idCidade),
  INDEX fk_Cidade_UF1_idx (idUF ASC),
  CONSTRAINT fk_Cidade_UF1
    FOREIGN KEY (idUF)
    REFERENCES UF (idUF)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table Bairro
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS Bairro (
  idBairro INT NOT NULL,
  dscBairro VARCHAR(45) NOT NULL,
  idCidade INT NOT NULL,
  PRIMARY KEY (idBairro),
  INDEX fk_Bairro_Cidade1_idx (idCidade ASC),
  CONSTRAINT fk_Bairro_Cidade1
    FOREIGN KEY (idCidade)
    REFERENCES Cidade (idCidade)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB;


-- -----------------------------------------------------
-- Table doador
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS doador (
  codDoador INT NOT NULL COMMENT 'Código do Doador',
  idTpSang INT NOT NULL COMMENT 'Código do Tipo do Doador - FK',
  nomDoador VARCHAR(40) NOT NULL COMMENT 'Nome do Doador',
  idBairro INT NOT NULL COMMENT 'Código do Bairro - FK',
  PRIMARY KEY (codDoador),
  INDEX fk_doador_TipoSanguineo1 (idTpSang ASC),
  INDEX fk_doador_Bairro1_idx (idBairro ASC),
  CONSTRAINT fk_doador_TipoSanguineo1
    FOREIGN KEY (idTpSang)
    REFERENCES TpSang (idTpSang),
  CONSTRAINT fk_doador_Bairro1
    FOREIGN KEY (idBairro)
    REFERENCES Bairro (idBairro)
    ON DELETE NO ACTION
    ON UPDATE NO ACTION)
ENGINE = InnoDB
DEFAULT CHARACTER SET = UTF8MB4;


-- -----------------------------------------------------
-- Table doacao
-- -----------------------------------------------------
CREATE TABLE IF NOT EXISTS doacao (
  codDoacao INT NOT NULL AUTO_INCREMENT,
  codDoador INT NOT NULL,
  datDoacao DATE NULL DEFAULT NULL,
  qtdSanDoacao INT NULL DEFAULT NULL,
  PRIMARY KEY (codDoacao),
  INDEX CODDOADOR (codDoador ASC),
  CONSTRAINT doacao_ibfk_1
    FOREIGN KEY (codDoador)
    REFERENCES doador (codDoador))
ENGINE = InnoDB
AUTO_INCREMENT = 15
DEFAULT CHARACTER SET = UTF8MB4;
